    let currentCategory = 'all';

    let isFlipped = false;

    let currentCard = null;


    function setCategory(cat) {

      currentCategory = cat;

      const btns = ['all', 'creative', 'mind'];

      btns.forEach(b => {

        const el = document.getElementById(`btn-${b}`);

        if(b === cat) {

          el.className = "px-4 py-1.5 rounded-full bg-amber-500 text-slate-950 font-bold transition";

        } else {

          el.className = "px-4 py-1.5 rounded-full text-slate-300 hover:text-amber-300 transition";

        }

      });

      if(isFlipped) resetCard();

    }


    function flipCard() {

      const cardEl = document.getElementById('card');

      if (!isFlipped) {

        pickRandomCard();

        cardEl.classList.add('flipped');

        isFlipped = true;


        confetti({

          particleCount: 40,

          spread: 60,

          origin: { y: 0.6 },

          colors: ['#f59e0b', '#fbbf24', '#d97706']

        });

      }

    }


    function resetCard() {

      const cardEl = document.getElementById('card');

      cardEl.classList.remove('flipped');

      isFlipped = false;

    }


    function pickRandomCard() {

      let pool = cards;

      if (currentCategory !== 'all') {

        pool = cards.filter(c => c.type === currentCategory);

      }

      const random = pool[Math.floor(Math.random() * pool.length)];

      currentCard = random;


      document.getElementById('cardCategory').textContent = random.tag;

      document.getElementById('cardId').textContent = `#${random.id}`;

      document.getElementById('cardQuote').textContent = random.quote;

      document.getElementById('cardDetail').textContent = random.detail;

    }


    function resetAndDraw() {

      const cardEl = document.getElementById('card');

      if (isFlipped) {

        cardEl.classList.remove('flipped');

        isFlipped = false;

        setTimeout(() => {

          flipCard();

        }, 400);

      } else {

        flipCard();

      }

    }


    function copyQuote(e) {

      e.stopPropagation();

      if (currentCard) {

        navigator.clipboard.writeText(`${currentCard.quote} — 微光靈感卡`);

        alert('✨ 已複製金句到剪貼簿！');

      }

    }
